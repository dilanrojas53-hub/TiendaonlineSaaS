    return [...map.entries()].map(([name, hex]) => ({ name, hex }));
  }, [products]);
  const sizes = useMemo(
    () => [...new Set(products.flatMap((product) => product.sizes))],
    [products],
  );

  const filtered = useMemo(() => {
    const query = search.trim().toLowerCase();
    const list = products.filter((product) => {
      const matchQuery =
        !query ||
        `${product.name} ${product.brand} ${product.category}`
          .toLowerCase()
          .includes(query);
      const matchCategory =
        category === "Todos" || product.category === category;
      const matchPrice = priceOf(product) <= maxPrice;
      const matchSize = !size || product.sizes.includes(size);
      const matchColor =
        !color || product.colors.some((item) => item.name === color);
      return (
        matchQuery && matchCategory && matchPrice && matchSize && matchColor
      );
    });
    return [...list].sort((a, b) => {
      if (sort === "Precio menor") return priceOf(a) - priceOf(b);
      if (sort === "Precio mayor") return priceOf(b) - priceOf(a);
      if (sort === "Novedades")
        return (
          Number(b.isNew) - Number(a.isNew) ||
          b.createdAt.localeCompare(a.createdAt)
        );
      return b.sold - a.sold;
    });
  }, [products, search, category, maxPrice, size, color, sort]);

  const cartLines = useMemo(

  const addToCart = () => {
    if (!selected) return;
    if (!selectedSize || !selectedColor) return flash("Elegí talla y color");
    if (selected.stock < 1) return flash("Producto agotado");
    setCart((current) => {
      const index = current.findIndex(
        (item) =>
          item.productId === selected.id &&
          item.size === selectedSize &&
          item.color === selectedColor,
      );
      if (index < 0)
        return [
          ...current,
          {
            productId: selected.id,
            size: selectedSize,
            color: selectedColor,
            quantity: 1,
          },
        ];
      return current.map((item, position) =>
        position === index
          ? { ...item, quantity: Math.min(selected.stock, item.quantity + 1) }
          : item,
      );
    });

    setCartOpen(true);
    flash("Agregado al carrito");

    setCart((current) =>
      current.flatMap((item, position) => {
        if (position !== index) return [item];
        const product = products.find(
          (candidate) => candidate.id === item.productId,
        );
        const next = Math.min(product?.stock || 0, item.quantity + delta);
        return next > 0 ? [{ ...item, quantity: next }] : [];
      }),
    );
